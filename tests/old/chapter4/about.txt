# This assumes that the current working directory is "MyProject"

import gettext
from pathlib import Path

# get a translation object
tr = gettext.translation("MyProject", Path("i18n/locale"), fallback=True)

# return translation object's gettext method
_ = tr.gettext

print(_("Hello World!"))
